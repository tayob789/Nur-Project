## FIX 2 — CONNECT PRAYER STATE GLOBALLY

Currently prayers are tracked in local screen state so HomeScreen and PrayersScreen are out of sync.

**Create a global prayer state using React Context:**

```typescript
// src/context/PrayerContext.tsx
import React, { createContext, useContext, useState, useEffect } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';

interface PrayerLog {
  name: string;
  done: boolean;
  doneAt: string | null;
}

interface PrayerContextType {
  todaysPrayers: PrayerLog[];
  togglePrayer: (name: string) => void;
  completedCount: number;
  streak: number;
}

const PrayerContext = createContext<PrayerContextType | null>(null);

const PRAYER_NAMES = ['Fajr', 'Dhuhr', 'Asr', 'Maghrib', 'Isha'];

export function PrayerProvider({ children }: { children: React.ReactNode }) {
  const today = new Date().toISOString().split('T')[0]; // YYYY-MM-DD
  const storageKey = `prayers_${today}`;

  const [todaysPrayers, setTodaysPrayers] = useState<PrayerLog[]>(
    PRAYER_NAMES.map(name => ({ name, done: false, doneAt: null }))
  );
  const [streak, setStreak] = useState(0);

  // Load today's prayers from AsyncStorage on mount
  useEffect(() => {
    AsyncStorage.getItem(storageKey).then(val => {
      if (val) setTodaysPrayers(JSON.parse(val));
    });
    calculateStreak();
  }, []);

  // Save to AsyncStorage whenever prayers change
  useEffect(() => {
    AsyncStorage.setItem(storageKey, JSON.stringify(todaysPrayers));
  }, [todaysPrayers]);

  const togglePrayer = (name: string) => {
    setTodaysPrayers(prev =>
      prev.map(p =>
        p.name === name
          ? { ...p, done: !p.done, doneAt: !p.done ? new Date().toISOString() : null }
          : p
      )
    );
  };

  // Calculate streak: how many consecutive days had all 5 prayers completed
  const calculateStreak = async () => {
    let count = 0;
    const checkDate = new Date();
    checkDate.setDate(checkDate.getDate() - 1); // start from yesterday

    for (let i = 0; i < 365; i++) {
      const dateKey = `prayers_${checkDate.toISOString().split('T')[0]}`;
      const val = await AsyncStorage.getItem(dateKey);
      if (!val) break;
      const logs: PrayerLog[] = JSON.parse(val);
      const allDone = logs.every(p => p.done);
      if (!allDone) break;
      count++;
      checkDate.setDate(checkDate.getDate() - 1);
    }
    setStreak(count);
  };

  const completedCount = todaysPrayers.filter(p => p.done).length;

  return (
    <PrayerContext.Provider value={{ todaysPrayers, togglePrayer, completedCount, streak }}>
      {children}
    </PrayerContext.Provider>
  );
}

export const usePrayers = () => {
  const ctx = useContext(PrayerContext);
  if (!ctx) throw new Error('usePrayers must be used within PrayerProvider');
  return ctx;
};
```

```typescript
// src/utils/greeting.ts
export function getGreeting(): { arabic: string; english: string } {
  const hour = new Date().getHours();
  
  if (hour >= 4 && hour < 7)   return { arabic: 'صباح الخير', english: 'Good morning' };
  if (hour >= 7 && hour < 12)  return { arabic: 'أصبحنا وأصبح الملك لله', english: 'A blessed morning' };
  if (hour >= 12 && hour < 15) return { arabic: 'انتهى الصباح', english: 'Good afternoon' };
  if (hour >= 15 && hour < 18) return { arabic: 'مساء الخير', english: 'Good afternoon' };
  if (hour >= 18 && hour < 21) return { arabic: 'مساء النور', english: 'Good evening' };
  return { arabic: 'طاب ليلك', english: 'Good night' };
}
```

```typescript
const MECCA_LAT = 21.4225;
const MECCA_LNG = 39.8262;

function calculateQiblaAngle(userLat: number, userLng: number): number {
  const φ1 = userLat * (Math.PI / 180);
  const φ2 = MECCA_LAT * (Math.PI / 180);
  const Δλ = (MECCA_LNG - userLng) * (Math.PI / 180);
  const x = Math.cos(φ2) * Math.sin(Δλ);
  const y = Math.cos(φ1) * Math.sin(φ2) - Math.sin(φ1) * Math.cos(φ2) * Math.cos(Δλ);
  return ((Math.atan2(x, y) * (180 / Math.PI)) + 360) % 360;
}

function getDistanceToMecca(lat: number, lng: number): number {
  const R = 6371; // km
  const φ1 = lat * Math.PI / 180;
  const φ2 = MECCA_LAT * Math.PI / 180;
  const Δφ = (MECCA_LAT - lat) * Math.PI / 180;
  const Δλ = (MECCA_LNG - lng) * Math.PI / 180;
  const a = Math.sin(Δφ/2)**2 + Math.cos(φ1)*Math.cos(φ2)*Math.sin(Δλ/2)**2;
  return Math.round(R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)));
}
```

```typescript
import Animated, { useSharedValue, useAnimatedStyle, withSpring, withSequence, withTiming } from 'react-native-reanimated';

function AnimatedPrayerRow({ prayer, onToggle }) {
  const scale = useSharedValue(1);
  const animStyle = useAnimatedStyle(() => ({ transform: [{ scale: scale.value }] }));
  
  const handlePress = () => {
    scale.value = withSequence(
      withTiming(0.96, { duration: 80 }),
      withSpring(1, { damping: 12 })
    );
    onToggle(prayer.name);
  };
  
  return (
    <Animated.View style={animStyle}>
      <TouchableOpacity onPress={handlePress} activeOpacity={1}>
        {/* prayer row content */}
      </TouchableOpacity>
    </Animated.View>
  );
}
```
